set(DMASH_QUICKJS_DIR "" CACHE PATH "QuickJS 2026-06-04 verified source tree")
set(DMASH_QUICKJS_LIBRARY "" CACHE FILEPATH "Optional matching prebuilt QuickJS static library")
if(NOT DMASH_QUICKJS_DIR)
 include(FetchContent)
 FetchContent_Declare(dmash_quickjs_source
  URL https://bellard.org/quickjs/quickjs-2026-06-04.tar.xz
  URL_HASH SHA256=b376e839b322978313d929fd20663b11ba58b75df5a46c126dd19ea2fa70ad2a)
 FetchContent_MakeAvailable(dmash_quickjs_source)
 set(DMASH_QUICKJS_DIR "${dmash_quickjs_source_SOURCE_DIR}")
endif()
# Export only the public header. On Windows QuickJS's VERSION file otherwise
# shadows the C++ standard <version> header through case-insensitive lookup.
set(DMASH_QUICKJS_PUBLIC_INCLUDE "${CMAKE_CURRENT_BINARY_DIR}/quickjs-public")
file(MAKE_DIRECTORY "${DMASH_QUICKJS_PUBLIC_INCLUDE}")
configure_file("${DMASH_QUICKJS_DIR}/quickjs.h" "${DMASH_QUICKJS_PUBLIC_INCLUDE}/quickjs.h" COPYONLY)
if(DMASH_QUICKJS_LIBRARY)
 add_library(dmash_quickjs STATIC IMPORTED)
 set_target_properties(dmash_quickjs PROPERTIES IMPORTED_LOCATION "${DMASH_QUICKJS_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${DMASH_QUICKJS_PUBLIC_INCLUDE}")
else()
 add_library(dmash_quickjs STATIC "${DMASH_QUICKJS_DIR}/quickjs.c" "${DMASH_QUICKJS_DIR}/cutils.c" "${DMASH_QUICKJS_DIR}/libregexp.c" "${DMASH_QUICKJS_DIR}/libunicode.c" "${DMASH_QUICKJS_DIR}/dtoa.c")
 target_include_directories(dmash_quickjs PRIVATE "${DMASH_QUICKJS_DIR}" INTERFACE "${DMASH_QUICKJS_PUBLIC_INCLUDE}")
 target_compile_definitions(dmash_quickjs PRIVATE _GNU_SOURCE CONFIG_VERSION="2026-06-04")
 target_compile_options(dmash_quickjs PRIVATE $<$<COMPILE_LANG_AND_ID:C,GNU,Clang>:-O2>)
endif()
