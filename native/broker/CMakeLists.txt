cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(DmashCredentialBroker LANGUAGES CXX)
 include(CTest)
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Sql)
find_package(OpenSSL 3.5.9 EXACT REQUIRED COMPONENTS Crypto)
add_library(dmash_credential_broker STATIC root_codec.cpp control_journal.cpp credential_store.cpp broker_service.cpp)
target_include_directories(dmash_credential_broker PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(dmash_credential_broker PUBLIC Qt6::Core Qt6::Sql PRIVATE OpenSSL::Crypto)
# Deliberately no network actor, QtNetwork, Account or QuickJS dependency.
if(BUILD_TESTING AND NOT ANDROID)
 add_executable(dmash_broker_codec_test tests/root_codec_test.cpp)
 target_link_libraries(dmash_broker_codec_test PRIVATE dmash_credential_broker)
 target_compile_definitions(dmash_broker_codec_test PRIVATE BROKER_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/compatibility.json")
 add_test(NAME broker-root-compatibility COMMAND dmash_broker_codec_test)
 add_executable(dmash_broker_service_test tests/credential_service_test.cpp)
 target_link_libraries(dmash_broker_service_test PRIVATE dmash_credential_broker)
 target_compile_definitions(dmash_broker_service_test PRIVATE BROKER_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/compatibility.json")
 add_test(NAME broker-credential-service COMMAND dmash_broker_service_test)
 add_executable(dmash_broker_control_test tests/control_journal_test.cpp)
 target_link_libraries(dmash_broker_control_test PRIVATE dmash_credential_broker)
 target_compile_definitions(dmash_broker_control_test PRIVATE BROKER_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/compatibility.json")
 add_test(NAME broker-control-journal COMMAND dmash_broker_control_test)
endif()
