cmake_minimum_required(VERSION 3.22)
project(DmashDesktop VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Sql Network Widgets Test)
find_package(OpenSSL 3.5.9 EXACT REQUIRED COMPONENTS Crypto)
include(CTest)
add_subdirectory(../../credential_host credential-owner EXCLUDE_FROM_ALL)
add_library(dmash_desktop_ui STATIC session_client.cpp credential_client.cpp profile_key.cpp account_controller.cpp message_history.cpp messenger_window.cpp)
target_include_directories(dmash_desktop_ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(dmash_desktop_ui PUBLIC Qt6::Widgets Qt6::Network)
if(WIN32)
 target_link_libraries(dmash_desktop_ui PRIVATE dmash_windows_local)
 target_include_directories(dmash_desktop_ui PRIVATE ../../src)
 target_link_libraries(dmash_desktop_ui PRIVATE advapi32)
endif()
add_executable(dmash-desktop main.cpp)
target_link_libraries(dmash-desktop PRIVATE dmash_desktop_ui)
add_dependencies(dmash-desktop dmash-credential-host)
install(TARGETS dmash-desktop RUNTIME DESTINATION bin)
install(DIRECTORY ../licenses/ DESTINATION share/dmash/licenses)
install(FILES ../../account/PROVENANCE.json DESTINATION share/dmash/licenses/account)
install(FILES ../../account/vendor/wasm2c/LICENSE DESTINATION share/dmash/licenses/account RENAME WABT-LICENSE.txt)
install(FILES ../../account/README.md DESTINATION share/dmash/licenses/account RENAME ACCOUNT-PROVENANCE.md)
if(BUILD_TESTING)
 add_executable(dmash_desktop_ui_test tests/ui_test.cpp)
 target_link_libraries(dmash_desktop_ui_test PRIVATE dmash_desktop_ui Qt6::Test)
 add_dependencies(dmash_desktop_ui_test dmash-credential-host)
 target_compile_definitions(dmash_desktop_ui_test PRIVATE CREDENTIAL_HOST_BINARY="$<TARGET_FILE:dmash-credential-host>" ACCOUNT_UI_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/../../account/tests/account_profile.json" BROKER_UI_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/../../broker/tests/compatibility.json")
 add_test(NAME desktop-controls COMMAND dmash_desktop_ui_test)
 set_tests_properties(desktop-controls PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../.." TIMEOUT 120 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
