cmake_minimum_required(VERSION 3.22)
project(DmashAndroid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core CorePrivate Gui Network WebSockets Sql)
find_package(OpenSSL 3.5.9 REQUIRED COMPONENTS Crypto)
add_subdirectory(../.. common)
add_subdirectory(../../credential_host credential-owner)
file(GLOB_RECURSE DMASH_PACKAGE_INPUTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/app/src/main/*" "${CMAKE_CURRENT_SOURCE_DIR}/../licenses/*")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${DMASH_PACKAGE_INPUTS})
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/app/src/main/" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/package-source")
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/../licenses/" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/package-source/assets/licenses")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/package-source/assets/licenses/account")
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/../../account/PROVENANCE.json" "${CMAKE_CURRENT_SOURCE_DIR}/../../account/README.md" "${CMAKE_CURRENT_SOURCE_DIR}/../../account/vendor/wasm2c/LICENSE" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/package-source/assets/licenses/account")
qt_add_executable(dmash_credential_service MANUAL_FINALIZATION service/credential_entry.cpp)
target_link_libraries(dmash_credential_service PRIVATE dmash_credential_host_core Qt6::Gui Qt6::CorePrivate)
target_link_options(dmash_credential_service PRIVATE -Wl,-z,max-page-size=16384)
qt_finalize_executable(dmash_credential_service)
qt_add_executable(dmash_node_service MANUAL_FINALIZATION service/entry.cpp)
# Compile the existing daemon entry as a callable service entry. No copied actor.
add_library(dmash_android_host_entry OBJECT ../../src/main.cpp)
target_compile_definitions(dmash_android_host_entry PRIVATE DMASH_HOST_EXTERNAL_APP)
target_link_libraries(dmash_android_host_entry PRIVATE dmash_native_actor)
target_link_libraries(dmash_node_service PRIVATE dmash_android_host_entry dmash_native_actor Qt6::Gui Qt6::CorePrivate)
set_target_properties(dmash_node_service PROPERTIES QT_ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/package-source" QT_ANDROID_PACKAGE_NAME "org.dmash.client" QT_ANDROID_VERSION_CODE "1" QT_ANDROID_VERSION_NAME "0.1.0-dev" QT_ANDROID_MIN_SDK_VERSION 28 QT_ANDROID_TARGET_SDK_VERSION 36)
target_link_options(dmash_node_service PRIVATE -Wl,-z,max-page-size=16384)
set_property(TARGET dmash_node_service PROPERTY QT_ANDROID_EXTRA_LIBS "${OPENSSL_CRYPTO_LIBRARY};${OPENSSL_INCLUDE_DIR}/../lib/libssl_3.so;${CMAKE_CURRENT_BINARY_DIR}/libdmash_credential_service_${ANDROID_ABI}.so")
qt_finalize_executable(dmash_node_service)
