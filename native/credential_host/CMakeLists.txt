cmake_minimum_required(VERSION 3.22)
project(DmashCredentialHost LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
include(CTest)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Sql Network)
find_package(OpenSSL 3.5.9 EXACT REQUIRED COMPONENTS Crypto)
if(NOT TARGET dmash_credential_broker)
 add_subdirectory(../broker broker-owner)
endif()
if(NOT TARGET dmash_account_service)
 add_subdirectory(../account account-owner)
endif()
add_library(dmash_credential_node_miner STATIC node_miner.cpp)
target_include_directories(dmash_credential_node_miner PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(TARGET dmash_quickjs)
 set(DMASH_MINER_JS_TARGET dmash_quickjs)
else()
 set(DMASH_MINER_JS_TARGET dmash_account_quickjs)
endif()
target_link_libraries(dmash_credential_node_miner PUBLIC dmash_credential_broker PRIVATE ${DMASH_MINER_JS_TARGET} OpenSSL::Crypto)
qt_add_resources(dmash_credential_node_miner node_miner PREFIX /dmash-node-miner BASE "${CMAKE_CURRENT_SOURCE_DIR}/../../D-MASH PWA/not_messenger/js/vendor" FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../D-MASH PWA/not_messenger/js/vendor/blake3.min.js")
if(BUILD_TESTING AND NOT ANDROID)
 add_executable(dmash_credential_node_enable_test tests/node_enable_test.cpp)
 target_link_libraries(dmash_credential_node_enable_test PRIVATE dmash_credential_node_miner)
 target_compile_definitions(dmash_credential_node_enable_test PRIVATE ACCOUNT_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/../account/tests/account_profile.json")
 add_test(NAME credential-first-node-enable COMMAND dmash_credential_node_enable_test)
 set_tests_properties(credential-first-node-enable PROPERTIES TIMEOUT 180)
endif()
if(UNIX AND NOT ANDROID)
 target_link_libraries(dmash_credential_node_miner PRIVATE m dl pthread)
endif()
add_library(dmash_credential_host_core STATIC credential_host.cpp daemon_rpc.cpp)
set_target_properties(dmash_credential_host_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(dmash_credential_host_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(dmash_credential_host_core PUBLIC dmash_account_service dmash_credential_broker dmash_credential_node_miner Qt6::Network)
if(WIN32)
 if(NOT TARGET dmash_windows_local)
  add_library(dmash_windows_local STATIC ../src/os_windows.cpp)
  target_include_directories(dmash_windows_local PUBLIC ../src)
  target_link_libraries(dmash_windows_local PUBLIC advapi32)
 endif()
 target_link_libraries(dmash_credential_host_core PRIVATE dmash_windows_local)
endif()
if(NOT ANDROID)
 add_executable(dmash-credential-host main.cpp supervisor.cpp)
 set_target_properties(dmash-credential-host PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
 target_link_libraries(dmash-credential-host PRIVATE dmash_credential_host_core)
 if(WIN32)
  target_link_libraries(dmash-credential-host PRIVATE dmash_windows_local)
 endif()
 install(TARGETS dmash-credential-host RUNTIME DESTINATION bin)
endif()
if(BUILD_TESTING AND NOT ANDROID)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 set(DMASH_CREDENTIAL_NODE_HOST "" CACHE FILEPATH "Pinned separate actual Node daemon for process lifecycle gate")
 if(DMASH_CREDENTIAL_NODE_HOST)
  add_test(NAME credential-host-process COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/process_owner_test.py" "$<TARGET_FILE:dmash-credential-host>" "${DMASH_CREDENTIAL_NODE_HOST}")
 else()
  add_test(NAME credential-host-process COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/process_owner_test.py" "$<TARGET_FILE:dmash-credential-host>")
 endif()
 set_tests_properties(credential-host-process PROPERTIES TIMEOUT 240)
endif()
