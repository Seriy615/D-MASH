cmake_minimum_required(VERSION 3.22)
project(DmashNativeInbox LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Sql)
add_library(dmash_native_inbox STATIC inbox_sql_store.cpp)
target_include_directories(dmash_native_inbox PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(dmash_native_inbox PUBLIC Qt6::Core Qt6::Sql)
include(CTest)
if(BUILD_TESTING AND NOT ANDROID)
  add_library(dmash_native_inbox_fault_test_port STATIC inbox_sql_store.cpp)
  target_include_directories(dmash_native_inbox_fault_test_port PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(dmash_native_inbox_fault_test_port PUBLIC DMASH_INBOX_FAULT_TESTS)
  target_link_libraries(dmash_native_inbox_fault_test_port PUBLIC Qt6::Core Qt6::Sql)
  add_executable(dmash_inbox_test tests/inbox_sql_test.cpp)
  target_link_libraries(dmash_inbox_test PRIVATE dmash_native_inbox_fault_test_port)
  add_test(NAME native-inbox COMMAND dmash_inbox_test)
  add_executable(dmash_inbox_bridge tests/inbox_bridge.cpp)
  target_link_libraries(dmash_inbox_bridge PRIVATE dmash_native_inbox)
endif()
