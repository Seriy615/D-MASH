cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(DmashAccountOwner LANGUAGES C CXX)
 include(CTest)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core)
find_package(OpenSSL 3.5.9 EXACT REQUIRED COMPONENTS Crypto)
set(DMASH_LEGACY_WASM "${CMAKE_CURRENT_SOURCE_DIR}/../../D-MASH PWA/not_messenger/js/vendor/kyber768.wasm")
file(SHA256 "${DMASH_LEGACY_WASM}" DMASH_LEGACY_WASM_SHA)
if(NOT DMASH_LEGACY_WASM_SHA STREQUAL "e9b316e56f5d1eef1f44f28bca99bec16ec9cff69af4b17d318143512255b009")
 message(FATAL_ERROR "Existing Account Kyber WASM changed; regenerate/review adapter explicitly")
endif()
add_library(dmash_account_legacy_kem STATIC legacy_kyber.cpp legacy_kyber_bridge.c vendor/kyber.c vendor/wasm2c/wasm-rt-impl.c)
set_target_properties(dmash_account_legacy_kem PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(dmash_account_legacy_kem PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE vendor vendor/wasm2c)
target_compile_definitions(dmash_account_legacy_kem PRIVATE WASM_RT_USE_MMAP=0 WASM_RT_MEMCHECK_BOUNDS_CHECK=1 WASM_RT_USE_STACK_DEPTH_COUNT=1)
target_link_libraries(dmash_account_legacy_kem PUBLIC Qt6::Core PRIVATE OpenSSL::Crypto)
if(UNIX)
 target_link_libraries(dmash_account_legacy_kem PRIVATE m)
endif()
if(BUILD_TESTING AND NOT ANDROID)
 add_executable(dmash_account_kem_test tests/legacy_kyber_test.cpp)
 target_link_libraries(dmash_account_kem_test PRIVATE dmash_account_legacy_kem)
 add_test(NAME account-legacy-kem-native COMMAND dmash_account_kem_test)
 set(DMASH_ACCOUNT_TEST_NODE "" CACHE FILEPATH "Node24 for exact existing JS/WASM Account KEM interop")
 if(DMASH_ACCOUNT_TEST_NODE)
  add_test(NAME account-legacy-kem-js COMMAND "${DMASH_ACCOUNT_TEST_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/legacy_kyber_interop.cjs" "$<TARGET_FILE:dmash_account_kem_test>" "${DMASH_LEGACY_WASM}")
 endif()
endif()

option(DMASH_BUILD_ACCOUNT_SERVICE "Build Account owner service with existing JS modules" ON)
if(DMASH_BUILD_ACCOUNT_SERVICE)
 find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Sql)
 if(TARGET dmash_quickjs)
  set(DMASH_ACCOUNT_JS_TARGET dmash_quickjs)
 else()
  set(DMASH_ACCOUNT_QUICKJS_DIR "" CACHE PATH "Verified QuickJS2026-06-04 source tree")
  if(NOT DMASH_ACCOUNT_QUICKJS_DIR)
   message(FATAL_ERROR "Set DMASH_ACCOUNT_QUICKJS_DIR to verified QuickJS2026-06-04 source")
  endif()
  add_library(dmash_account_quickjs STATIC "${DMASH_ACCOUNT_QUICKJS_DIR}/quickjs.c" "${DMASH_ACCOUNT_QUICKJS_DIR}/cutils.c" "${DMASH_ACCOUNT_QUICKJS_DIR}/libregexp.c" "${DMASH_ACCOUNT_QUICKJS_DIR}/libunicode.c" "${DMASH_ACCOUNT_QUICKJS_DIR}/dtoa.c")
  set_target_properties(dmash_account_quickjs PROPERTIES POSITION_INDEPENDENT_CODE ON)
  # Export only quickjs.h: the source VERSION file shadows C++ <version>
  # on case-insensitive Windows filesystems. Preserve the verified source tree.
  set(DMASH_ACCOUNT_QUICKJS_PUBLIC_INCLUDE "${CMAKE_CURRENT_BINARY_DIR}/quickjs-public")
  file(MAKE_DIRECTORY "${DMASH_ACCOUNT_QUICKJS_PUBLIC_INCLUDE}")
  configure_file("${DMASH_ACCOUNT_QUICKJS_DIR}/quickjs.h" "${DMASH_ACCOUNT_QUICKJS_PUBLIC_INCLUDE}/quickjs.h" COPYONLY)
  target_include_directories(dmash_account_quickjs PRIVATE "${DMASH_ACCOUNT_QUICKJS_DIR}" INTERFACE "${DMASH_ACCOUNT_QUICKJS_PUBLIC_INCLUDE}")
  target_compile_definitions(dmash_account_quickjs PRIVATE _GNU_SOURCE CONFIG_VERSION="2026-06-04")
  target_compile_options(dmash_account_quickjs PRIVATE $<$<COMPILE_LANG_AND_ID:C,GNU,Clang>:-O2>)
  set(DMASH_ACCOUNT_JS_TARGET dmash_account_quickjs)
 endif()
 add_library(dmash_account_service STATIC account_service.cpp account_sql_store.cpp ../src/crypto_port.cpp)
 set_target_properties(dmash_account_service PROPERTIES POSITION_INDEPENDENT_CODE ON)
 target_include_directories(dmash_account_service PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(dmash_account_service PUBLIC dmash_account_legacy_kem Qt6::Core Qt6::Sql PRIVATE ${DMASH_ACCOUNT_JS_TARGET} OpenSSL::Crypto)
 if(UNIX AND NOT ANDROID)
  target_link_libraries(dmash_account_service PRIVATE m dl pthread)
 endif()
 include(js/resources.cmake)
 if(BUILD_TESTING AND NOT ANDROID)
  add_executable(dmash_account_service_test tests/account_service_test.cpp)
  target_compile_definitions(dmash_account_service_test PRIVATE ACCOUNT_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/account_profile.json")

   add_subdirectory(../broker broker-for-account)
  endif()
  target_link_libraries(dmash_account_service_test PRIVATE dmash_account_service dmash_credential_broker)
  add_executable(dmash_account_private_probe tests/private_profile_probe.cpp)
  target_link_libraries(dmash_account_private_probe PRIVATE dmash_account_service dmash_credential_broker OpenSSL::Crypto)
  add_test(NAME account-service-retained-profile COMMAND dmash_account_service_test)
  add_executable(dmash_account_import_recovery_test tests/import_recovery_test.cpp)
  target_compile_definitions(dmash_account_import_recovery_test PRIVATE ACCOUNT_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/account_profile.json")
  target_link_libraries(dmash_account_import_recovery_test PRIVATE dmash_account_service dmash_credential_broker)
  add_test(NAME account-import-recovery COMMAND dmash_account_import_recovery_test)
  find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Network)
  add_executable(dmash_account_remote_probe tests/native_remote_probe.cpp ../credential_host/daemon_rpc.cpp)
  target_include_directories(dmash_account_remote_probe PRIVATE ../credential_host)
  target_link_libraries(dmash_account_remote_probe PRIVATE dmash_account_service dmash_credential_broker Qt6::Network)
 endif()
endif()
