cmake_minimum_required(VERSION 3.22)
project(DmashToolchainProbe LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(QJS_DIR "" CACHE PATH "Verified upstream QuickJS source directory")
if(NOT EXISTS "${QJS_DIR}/quickjs.c")
 message(FATAL_ERROR "Set QJS_DIR to verified QuickJS 2026-06-04 sources")
endif()
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Network WebSockets Sql)
add_library(qjs STATIC "${QJS_DIR}/quickjs.c" "${QJS_DIR}/cutils.c" "${QJS_DIR}/libregexp.c" "${QJS_DIR}/libunicode.c" "${QJS_DIR}/dtoa.c")
set_target_properties(qjs PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(qjs PRIVATE _GNU_SOURCE CONFIG_VERSION="2026-06-04")
target_include_directories(qjs PUBLIC "${QJS_DIR}")
if(ANDROID)
 qt_add_library(dmash_toolchain_service SHARED android_main.cpp)
 target_link_options(dmash_toolchain_service PRIVATE "LINKER:-z,max-page-size=16384")
 target_link_libraries(dmash_toolchain_service PRIVATE qjs Qt6::Core Qt6::Network Qt6::WebSockets Qt6::Sql log)
else()
 find_package(OpenSSL REQUIRED)
 add_executable(dmash_toolchain_probe desktop_main.cpp)
 target_link_libraries(dmash_toolchain_probe PRIVATE qjs Qt6::Core Qt6::Network Qt6::WebSockets Qt6::Sql OpenSSL::Crypto)
endif()
